Security-Context
-----------------

root@controlplane:~$ k run pod --image=busybox --command -o yaml --dry-run=client > pod.yml -- sh -c 'sleep 1d'
root@controlplane:~$ ls
filesystem  pod.yml
root@controlplane:~$ cat pod.yml 
apiVersion: v1
kind: Pod
metadata:
  labels:
    run: pod
  name: pod
spec:
  containers:
  - command:
    - sh
    - -c
    - sleep 1d
    image: busybox
    name: pod
    resources: {}
  dnsPolicy: ClusterFirst
  restartPolicy: Always
status: {}
root@controlplane:~$ k apply -f pod.yml 
pod/pod created
root@controlplane:~$ k get po 
NAME   READY   STATUS              RESTARTS   AGE
pod    0/1     ContainerCreating   0          3s
root@controlplane:~$ k exec -it pod -- sh
/ # id
uid=0(root) gid=0(root) groups=0(root),10(wheel)
/ # exit
root@controlplane:~$ vi pod.yml 
root@controlplane:~$ cat pod.yml 
apiVersion: v1
kind: Pod
metadata:
  labels:
    run: pod
  name: pod
spec:
  securityContext:
    runAsUser: 1000
    runAsGroup: 3000
  containers:
  - command:
    - sh
    - -c
    - sleep 1d
    image: busybox
    name: pod
    resources: {}
  dnsPolicy: ClusterFirst
  restartPolicy: Always
status: {}
root@controlplane:~$ k apply -f pod.yml --force
pod/pod created
