Secrets
-------

2 ways we can attach secrets to pod
  1. env vars
  2. mount as volumes

root@controlplane:~$ k create secret generic secret1 --from-literal user=admin
secret/secret1 created
root@controlplane:~$ k create secret generic secret2 --from-literal password=12345678
secret/secret2 created
root@controlplane:~$ k run pod --image=nginx -o yaml --dry-run=client > pod.yml
root@controlplane:~$ ls
filesystem  pod.yml
root@controlplane:~$ vi pod.yml
root@controlplane:~$ cat pod.yml 
apiVersion: v1
kind: Pod
metadata:
  labels:
    run: pod
  name: pod
spec:
  volumes:
    - name: secret-volume
      secret:
        secretName: secret1
  containers:
  - image: nginx
    name: pod
    env:
    - name: password
      valueFrom:
        secretKeyRef:
          name: secret2
          key: password
    volumeMounts:
    - name: secret-volume
      readOnly: true
      mountPath: "/etc/secret1"
    resources: {}

root@controlplane:~$ k apply -f pod.yml --force
pod/pod created
