Why disable automount? [/var/run/secrets/kubernetes.io/serviceaccount/token]
if its ServiceAccount token is mounted, a compromised application might use that token to attempt API access.
Suppose you deploy an NGINX application -- Does NGINX need to communicate with the Kubernetes API server? -- Usually, no.

EX:
---
root@controlplane:~$ vi nginx.yml
apiVersion: v1
kind: Pod
metadata:
  name: nginx
spec:
  automountServiceAccountToken: false
  serviceAccountName: k8s
  containers:
  - name: nginx
    image: nginx
root@controlplane:~$ k apply -f nginx.yml 
pod/nginx created
root@controlplane:~$ k exec -it nginx -- bash
root@nginx:/# pwd
/
root@nginx:/# mount | grep service
