#########lets scan an k8s image
root@controlplane:~$ k -n kube-system get pod | grep api
kube-apiserver-controlplane            1/1     Running   2 (36m ago)   22d
root@controlplane:~$ k -n kube-system get pod kube-apiserver-controlplane -o yaml | grep image
    image: registry.k8s.io/kube-apiserver:v1.37.1
    imagePullPolicy: IfNotPresent
    image: registry.k8s.io/kube-apiserver:v1.37.1
    imageID: registry.k8s.io/kube-apiserver@sha256:e190f914a6cc21cab9268485d1ae7282fdeab1315ac64b668aa42d732fd1061d
root@controlplane:~$ docker run ghcr.io/aquasecurity/trivy:latest image registry.k8s.io/kube-apiserver:v1.37.1

root@controlplane:~$ kubectl get pods -A -o yaml | grep "image:"
root@controlplane:~$ cat /etc/kubernetes/manifests/kube-apiserver.yaml | grep image
    image: registry.k8s.io/kube-apiserver:v1.37.1
    imagePullPolicy: IfNotPresent

####this is the version of the image which is running in your cluster for api server lets change the image id with an sha value of the image 
##### and see that the api server come back or not
root@controlplane:~$ sudo sed -i 's|registry.k8s.io/kube-apiserver:v1.37.1|registry.k8s.io/kube-apiserver@sha256:e190f914a6cc21cab9268485d1ae7282fdeab1315ac64b668aa42d732fd1061d|' /etc/kubernetes/manifests/kube-apiserver.yaml
######this way no one will know which version of the image we are running wait for few minutes till the kube-api server comes back
