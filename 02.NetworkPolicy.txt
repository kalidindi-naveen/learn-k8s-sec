NetworkPolicies: A set of rules that define which network traffic is allowed or denied for Pods.
CNI (Calico): Implements NetworkPolicies using iptables or eBPF to allow or block traffic between Pods, including Pods across different namespaces.

### Block all traffic within namespace
root@controlplane:~$ vi deny.yml
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: default-deny
  namespace: default
spec:
  podSelector: {}
  policyTypes:
  - Egress
  - Ingress

root@controlplane:~$ k apply -f deny.yml 
networkpolicy.networking.k8s.io/default-deny created

apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: fe-be
  namespace: default
spec:
  podSelector:
    matchLabels:
      run: fe
  policyTypes:
  - Egress
  egress:
  - to:
    - podSelector:

          run: be

Note: Need to allow DNS by modifying defaul-deny.yml
root@controlplane:~$ vi deny.yml
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: default-deny
  namespace: default
spec:
  podSelector: {}
  policyTypes:
  - Egress
  - Ingress
  egress:
  - to:
    ports:
    - protocol: TCP
      port: 53
    - protocol: UDP
      port: 53

root@controlplane:~$ k apply -f deny.yml 
networkpolicy.networking.k8s.io/default-deny configured

# Works
root@controlplane:~$ k exec -it fe -- curl be
