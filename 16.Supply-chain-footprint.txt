Container Best Practicies
-------------------------
root@controlplane:~$ vi Dockerfile
root@controlplane:~$ cat Dockerfile 
FROM ubuntu
ARG DEBIAN_FRONTEND=noninteractive
RUN apt-get update && apt-get install -y golang-go
COPY app.go .
RUN CGO_ENABLED=0 go build app.go
CMD ["./app"]
root@controlplane:~$ vi app.go
root@controlplane:~$ cat app.go 
package main

import (
    "fmt"
    "time"
    "os/user"
)

func main () {
    user, err := user.Current()
    if err != nil {
        panic(err)
    }

    for {
        fmt.Println("user: " + user.Username + " id: " + user.Uid)
        time.Sleep(1 * time.Second)
    }
}
root@controlplane:~$ docker build -t app .
root@controlplane:~$ docker run app
user: root id: 0

root@controlplane:~$ mkdir non-root
root@controlplane:~$ cd non-root/
root@controlplane:~/non-root$ vi Dockerfile
root@controlplane:~/non-root$ cat Dockerfile 
FROM ubuntu
ARG DEBIAN_FRONTEND=noninteractive
RUN apt-get update && apt-get install -y golang-go
COPY app.go .
RUN CGO_ENABLED=0 go build app.go

FROM alpine
RUN addgroup -S appgroup && adduser -S appuser -G appgroup -h /home/appuser
COPY --from=0 /app /home/appuser/
USER appuser
CMD ["/home/appuser/app"]
root@controlplane:~/non-root$ cp ../app.go .
root@controlplane:~/non-root$ ls
Dockerfile  app.go
root@controlplane:~/non-root$ docker build -t app3 .
root@controlplane:~/non-root$ docker image list | grep app3
WARNING: This output is designed for human readability. For machine-readable output, please use --format.

root@controlplane:~/non-root$ docker run app3
user: appuser id: 100
