CSR
---
Certificate Request Token
https://kubernetes.io/docs/tasks/tls/certificate-issue-client-csr/

Hands-on:
---------
1. create cert+key and authenticate as user naveen
2. will create csr
3. will sign csr using k8s api [importing csr]
4. using cert+key will connect to k8s api as user jane

Steps:
------
Generate TLS Certificate [we are using openssl]
root@controlplane:~$ openssl genrsa -out myuser.key 3072
root@controlplane:~$ ls
filesystem  myuser.key

root@controlplane:~$ openssl req -new -key myuser.key -out myuser.csr -subj "/CN=myuser"
root@controlplane:~$ ls
filesystem  myuser.csr  myuser.key

root@controlplane:~$ cat myuser.csr | base64 -w 0

root@controlplane:~$ cat <<EOF | kubectl apply -f -
apiVersion: certificates.k8s.io/v1
kind: CertificateSigningRequest
metadata:
  name: myuser
spec:
  request: <base64-encoded-csr>
  signerName: kubernetes.io/kube-apiserver-client
  expirationSeconds: 86400  # one day
  usages:
  - client auth
EOF
certificatesigningrequest.certificates.k8s.io/myuser created

root@controlplane:~$ kubectl get csr | grep myuser
myuser      39s   kubernetes.io/kube-apiserver-client           kubernetes-admin           24h                 Pending

root@controlplane:~$ kubectl certificate approve myuser
certificatesigningrequest.certificates.k8s.io/myuser approved

root@controlplane:~$ kubectl get csr myuser -o jsonpath='{.status.certificate}'| base64 -d > myuser.crt
root@controlplane:~$ ls
filesystem  myuser.crt  myuser.csr  myuser.key

root@controlplane:~$ k config view
