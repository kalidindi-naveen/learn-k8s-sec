Kubernetes does not enable audit logging by default in a standard kubeadm cluster.

root@controlplane:~$ cd /etc/kubernetes/
root@controlplane:/etc/kubernetes$ mkdir audit
root@controlplane:/etc/kubernetes$ cd audit
root@controlplane:/etc/kubernetes$ vi policy.yaml
root@controlplane:/etc/kubernetes$ cat policy.yaml 
apiVersion: audit.k8s.io/v1 
kind: Policy
rules:
  - level: Metadata
root@controlplane:/etc/kubernetes$ vi manifests/kube-apiserver.yaml 
##below the contianer defination
spec:
  containers:
  - command:
    - kube-apiserver
    - --audit-policy-file=/etc/kubernetes/audit/policy.yaml ##this is the path of your policy
    - --audit-log-path=/etc/kubernetes/audit/logs/audit.log #log path
    - --audit-log-maxbackup=5 #number of logs
    - --audit-log-maxsize=500 #size of the logs
###################################################
lets add the volume in the volume block
 volumes:
   - hostPath:
      path: /etc/kubernetes/audit
      type: DirectoryOrCreate
    name: audit
  
##lets add the mount path in the container block
- mountPath: /etc/kubernetes/audit
  name: audit
