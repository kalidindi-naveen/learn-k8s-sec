### Setup Ingress
root@controlplane:~$ k run fe --image nginx
pod/fe created
root@controlplane:~$ k expose po fe --port 80
service/fe exposed

root@controlplane:~$ vi ing.yml 
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: minimal-ingress
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /  # We need this annotation because your backend NGINX serves the application at /, but the user accesses it through /fe.
spec:
  ingressClassName: nginx
  rules:
  - http:
      paths:
      - path: /fe
        pathType: Prefix
        backend:
          service:
            name: fe
            port:
              number: 80

root@controlplane:~$ k apply -f ing.yml 
ingress.networking.k8s.io/minimal-ingress created

root@controlplane:~$ k get ingress
